#用来筛选文件夹中的文件是否以Excel表中某个特定范围内的格子内容为名，
#且结尾为指定内容结尾，剪切到另一文件夹下
import openpyxl
import os
import shutil
#选定Excel表格
workbook = openpyxl.load_workbook('data.xlsx')
#选定工作表
worksheet = workbook['E_CODE_Train']
#选定读取范围
read_range=worksheet['G2':'G2103']
#先创建一个空数组，以后用来存放正确的文件名
right_name=[]
for cell in read_range:
    for c in cell:
        if(c.value!='Null'):
            #print(worksheet.cell(row=c.row,column=1).value)
            #将G列不为Null那一行的A列数据写入数组
            right_name.append(worksheet.cell(row=c.row,column=1).value)
#原文件夹
raw_picture_folder_path = 'C:\\Users\\rolen\\Downloads\\HIT视觉POC\\raw_picture'
#目标文件夹
targat_folder_path ='C:\\Users\\rolen\\Downloads\\HIT视觉POC\\target_picture'
#另外的筛选条件
filter_1=['RT.jpg','TP.jpg','RR.jpg','LT.jpg','FR.jpg']
#遍历这个文件夹下的文件
for file in os.listdir(raw_picture_folder_path):
    #如果文件名开头在数组里且结尾是给定的几个结尾之一
    if any(file.startswith(prefix) and file.endswith(endfix) for prefix in right_name for endfix in filter_1):
        ex_file = os.path.join(raw_picture_folder_path,file)
        now_file = os.path.join(targat_folder_path,file)
        shutil.move(ex_file,now_file)
