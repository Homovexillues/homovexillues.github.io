import os
import requests
from bs4 import BeautifulSoup
from PIL import Image
from io import BytesIO
#这个可以用来爬一般的有图片的静态网站

def download_images(url,save_path,suffix =1):
    os.makedirs(save_path, exist_ok=True)
    response = requests.get(url)
    soup = BeautifulSoup(response.text,'html.parser')
    image_tags = soup.find_all('img')
    if not os.path.exists(save_path):
        os.mkdir(save_path)

    for img in image_tags:
        img_url = img.get('src')
        if 'https' not in img_url:
            img_url = url+img_url
        img_data = requests.get(img_url).content
        #使用合法的文件名
        img_name = os.path.join(save_path,os.path.basename(img_url).split('?')[0])
        img_name, ext = os.path.splitext(img_name)
        img_name = os.path.join(save_path,img_name)

      

        if os.path.exists(os.path.join(save_path,img_name)+".jpg"):
            while os.path.exists(os.path.join(save_path,f"{img_name}_{suffix}.jpg")):
                suffix+=1
            img_name = f"{img_name}_{suffix}"
        print(img_name)
        #将图片转换为.jpg格式并保存
        try:
            img = Image.open(BytesIO(img_data))
        except IOError:
            print(f"Cannot identify image file from URL {img_url}")
        else:
            if ext == '.webp':
                # 如果原始格式是 .webp，将其强制转换为 .jpg 格式
                img.save(img_name + ".jpg", 'JPEG')
            else:
                # 否则，保持原始格式
                img.save(img_name + ext)
