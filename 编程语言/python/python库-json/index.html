#用loads()函数读取json
>>>stringofjsonData ='{"name":"Zophie","isCate":true,"miceCaught":0,"felineIQ":null}'
>>>import json
>>>jsonDataAsPythonValue = json.loads(stringofjsonData)
>>>jsonDataAsPythonValue
{'name': 'Zophie', 'isCate': True, 'miceCaught': 0, 'felineIQ': None}
#loads()函数会将数据返回为一个Python字典，但是python字典是没有顺序的，因此打印jsonDataAsPythonValue可能会以不同的顺序出现
>>>data = '''
[{
    "name": "小明",
    "height": "170",
    "age": "18"
}, {
     "name": "小红",
    "height": "165",
    "age": "20"
}]
'''
>>>new_data=json.loads(data)

import json
data = '''
[{
    "name": "小明",
    "height": "170",
    "age": "18"
}, {
     "name": "小红",
    "height": "165",
    "age": "20"
}]
'''
new_data = json.loads(data)

name_1 = new_data[0]['name']
name_2 = new_data[1]['name']
print(name_1)
print(name_2)
