import os
import re

files = os.listdir('.')
for file in files:
    if file.endswith('.mp3.lrc'):
        new_file = re.sub(r'\.mp3\.lrc$','.lrc',file)
        os.rename(file,new_file)
    elif file.endswith('.mp3.vtt'):#如果发现是vtt格式则自动修改为lrc格式
        new_file = re.sub(r'\.mp3\.vtt$','.vtt',file)
        os.rename(file,new_file)
        vtt_to_lrc(new_file)
    elif file.endswith('.wav.vtt'):
        new_file = re.sub(r'\.wav\.vtt$','.vtt',file)
        os.rename(file,new_file)
        vtt_to_lrc(new_file)

import os
import re

def vtt_to_lrc(vtt_file):
    # 打开 .vtt 文件
    with open(vtt_file, 'r',encoding='utf-8') as f:
        lines = f.readlines()

    # 准备 .lrc 文件内容
    lrc_content = ''

    # 遍历 .vtt 文件中的行
    for line in lines:
        # 如果该行是时间戳行
        if re.match(r'\d{2}:\d{2}:\d{2}.\d{3} --> \d{2}:\d{2}:\d{2}.\d{3}', line):
            # 将时间戳转换为 .lrc 格式
            start_time, end_time = line.split(' --> ')
            start_min, start_sec = start_time.split(':')[-2:]
            start_sec, start_msec = start_sec.split('.')
            lrc_timestamp = f'[{start_min}:{start_sec}.{start_msec}]'
            lrc_content += lrc_timestamp
        elif line.strip() and not line.startswith('WEBVTT') and not re.match(r'NOTE.*', line) and not re.match(r'\d+', line):
            # 如果该行是字幕行，将其添加到 .lrc 文件内容中
            lrc_content += line

    # 将 .lrc 文件内容写入新的 .lrc 文件
    lrc_file = vtt_file.replace('.vtt', '.lrc')
    with open(lrc_file, 'w',encoding='utf-8') as f:
        f.write(lrc_content)

    print(f"{vtt_file} 已成功转换为 {lrc_file}.")
