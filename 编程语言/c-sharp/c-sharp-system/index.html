using System;
using System.ComponentModel;

// 以下示例演示了如何创建一个实现 IDisposable 接口
// 和 IDisposable.Dispose 方法的资源类。

public class DisposeExample
{
    // 实现 IDisposable 的基类。
    // 通过实现 IDisposable，你在声明
    // 这种类型的实例分配了稀缺资源。
    public class MyResource: IDisposable
    {
        // 指向外部非托管资源的指针。
        private IntPtr handle;
        // 此类使用的其他托管资源。
        private Component component = new Component();
        // 跟踪是否已调用 Dispose。
        private bool disposed = false;

        // 类构造函数。
        public MyResource(IntPtr handle)
        {
            this.handle = handle;
        }

        // 实现 IDisposable。
        // 不要使此方法成为虚方法。
        // 派生类不应能够重写此方法。
        public void Dispose()
        {
            Dispose(disposing: true);
            // 此对象将由 Dispose 方法清理。
            // 因此，你应该调用 GC.SuppressFinalize 来
            // 将此对象从终结队列中取出
            // 并防止此对象的终结代码
            // 执行第二次。
            GC.SuppressFinalize(this);
        }

        // Dispose(bool disposing) 在两种不同的场景中执行。
        // 如果 disposing 等于 true，该方法已直接或间接
        // 由用户的代码调用。可以处置托管和非托管资源。
        // 如果 disposing 等于 false，该方法已由运行时从
        // 终结器内部调用，你不应引用其他对象。只能处置非托管资源。
        protected virtual void Dispose(bool disposing)
        {
            // 检查是否已调用 Dispose。
            if(!this.disposed)
            {
                // 如果 disposing 等于 true，处置所有托管
                // 和非托管资源。
                if(disposing)
                {
                    // 处置托管资源。
                    component.Dispose();
                }

                // 在此处调用适当的方法来清理
                // 非托管资源。
                // 如果 disposing 为 false，
                // 只执行以下代码。
                CloseHandle(handle);
                handle = IntPtr.Zero;

                // 注意已进行处置。
                disposed = true;
            }
        }

        // 使用互操作调用必要的方法
        // 来清理非托管资源。
        [System.Runtime.InteropServices.DllImport("Kernel32")]
        private extern static Boolean CloseHandle(IntPtr handle);

        // 使用 C# 终结器语法进行终结代码。
        // 此终结器只会在未调用 Dispose 方法时运行。
        // 它给予你的基类机会进行终结。
        // 不要在此类派生的类型中提供终结器。
        ~MyResource()
        {
            // 不要在此处重新创建 Dispose 清理代码。
            // 调用 Dispose(disposing: false) 在可读性和可维护性方面是最佳的。
            Dispose(disposing: false);
        }
    }
    public static void Main()
    {
        // 在此处插入代码以创建
        // 并使用 MyResource 对象。
    }
}
