using System;

public interface Itype
{
    public static string[]? allFiles { get; set; }
    public static string[]? txtFiles { get; set; }
    public static string[]? jsonFiles { get; set; }
    public static string[]? jpgFiles { get; set; }
}

public class Sort:Itype
{
    private static List<string> fileNames = new();

    private static Random rng = new();

    //向列表里添加内容
    public static void addList(string[] listName)
    {
        foreach (string fileName in listName)
        {
            var name = Path.GetFileName(fileName);
            fileNames.Add(name);
        }
    }

    //获取指定位置的所有文件名字添加到列表中
    public static void getName(string filePath)
    {
        
        Itype.allFiles = Directory.GetFiles(filePath);
        addList(Itype.allFiles);
    }

    //指定特定类型文档
    public static void getName(string filePath, string type)
    {
        
        Itype.txtFiles = Directory.GetFiles(filePath, "*.txt");
        Itype.jsonFiles = Directory.GetFiles(filePath, "*.json");
        Itype.jpgFiles = Directory.GetFiles(filePath, "*.jpg");
        switch (type)
        {
            case "txt": addList(Itype.txtFiles); break;
            case "json": addList(Itype.jsonFiles); break;
            case "jpg": addList(Itype.jpgFiles); break;
        }
    }

    //随机抽指定比例的文件移动到另一文件夹
    public static void moveFile(string sourcePath, string targetPath,double Rate)
    {
        getName(sourcePath);
        int length = (int)(fileNames.Count *(Rate));
        var selectedFiles = fileNames.OrderBy(x => rng.Next()).Take(length);
        foreach (var file in selectedFiles)
        {
            var sourceFile = Path.Combine(sourcePath, file);
            var targetFile = Path.Combine(targetPath, file);
            File.Move(sourceFile, targetFile);
        }
    }

    public static void Main()
    {
        //string UnSortPictureFilePath = @"C:\Users\rolen\Documents\Damage_picture（第二批）\";
        string UnSortPictureFilePath = @"C:\Users\rolen\Documents\Damage_picture（第二批）\train";
        //string SortedPictureFilePath = @"C:\uzone\项目\2024\2024-02 PoC视觉\文档\图像训练第二批（20240407）\images\train\";
        string SortedPictureFilePath = @"C:\Users\rolen\Documents\Damage_picture（第二批）\val";
