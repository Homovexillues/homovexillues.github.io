using System;
using System.IO;
using System.Text;

namespace FileApplication
{
    class Program
    {
        static void Main(string[] args)
        {
            try
            {
                //创建一个StreamReader的实例来读取文件
                //using语句也能关闭StreamReader
                using (StreamReader sr = new StreamReader("C:\\Users\\rolen\\Documents\\硬盘处理.md",Encoding.UTF8))
                {
                    string line;
                    //从文件中读取并显示行，直到文件的末尾
                    while((line = sr.ReadLine()) != null)
                    {
                        Console.WriteLine(line);
                    }
                }
            }
            catch (Exception e) 
            {
                //向用户显示处错信息
                Console.WriteLine ("The file could not be read0");
                Console.WriteLine (e.Message);
            }
            Console.ReadKey();
        }
    }
}

using System;
using System.IO;
using System.Text;

namespace FileApplication
{
    class Program
    {
        static void Main(string[] args)
        {
            string[] names = new string[] {"Zara Ali","Nuha Ali" };
            using (StreamWriter sw = new StreamWriter("C:\\Users\\rolen\\Documents\\test.txt"))
            {
                foreach (string name in names)
                {
                    sw.WriteLine(name);
                }
            }
            //从文件中读取并显示每行
            string line = "";
            using(StreamReader sr = new StreamReader("C:\\Users\\rolen\\Documents\\test.txt"))
            {
                while((line = sr.ReadLine())!= null)
                {
                    Console.WriteLine(line);
                }
            }
            Console.ReadKey();
        }
    }
}

using System;
using System.IO;
using System.Linq;
using System.Text.RegularExpressions;

class Program
{
    static void Main()
    {
        string target_path = "C:\\Users\\rolen\\Documents\\example\\DataSet-tangshan0805\\labels\\train";
        string File_path = "C:\\Users\\rolen\\Documents\\example\\log.txt";

        //遍历目录下的所有文件
        foreach (string filepath in Directory.GetFiles(target_path,"*.json"))//可以通过正则表达式指定文件类型
        {
            try
            {
                //读取文件内容
                string content = File.ReadAllText(filepath);

                //使用正则表达式找到第一个数字
                Match match = Regex.Match(content, @"\d+");
                if (match.Success)
                {
                    //将数字写入txt文档
                    File.AppendAllText(File_path, match.Value + Environment.NewLine);

                }
            }
            catch(Exception ex) 
            {
                Console.WriteLine($"处理文件{filepath}时出错:{ex.Message}");

            }
        }
    }
}
