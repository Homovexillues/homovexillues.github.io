def is_stationary(gyro_data, threshold=0.1, stable_time=1.0):
    """
    gyro_data: 角速度数据队列
    threshold: 静止阈值(如0.1°/s)
    stable_time: 稳定时间(如1秒)
    """
    # 检查连续数据是否都小于阈值
    for i in range(1, len(gyro_data)):
        delta = abs(gyro_data[i] - gyro_data[i-1])
        if delta > threshold:
            return False
    return True

class GyroCalibrator:
    def __init__(self, threshold=0.1, stable_time=1.0, sample_rate=100):
        self.threshold = threshold
        self.samples_needed = int(stable_time * sample_rate)
        self.bias = 0.0
        self.is_calibrated = False
        self.data_buffer = []
    
    def update(self, raw_gyro):
        # 如果已经校准，直接应用零偏补偿
        if self.is_calibrated:
            return raw_gyro - self.bias
        
        # 缓存数据
        self.data_buffer.append(raw_gyro)
        if len(self.data_buffer) > self.samples_needed:
            self.data_buffer.pop(0)
        
        # 检查是否可以校准
        if len(self.data_buffer) == self.samples_needed and self._check_stationary():
            self.bias = sum(self.data_buffer) / len(self.data_buffer)
            self.is_calibrated = True
            print(f"零偏校准完成: bias = {self.bias:.3f}°/s")
        
        return raw_gyro  # 返回未校准数据
    
    def _check_stationary(self):
        """检查传感器是否静止"""
        for i in range(1, len(self.data_buffer)):
            if abs(self.data_buffer[i] - self.data_buffer[i-1]) > self.threshold:
                return False
        return True

# 使用示例

for t in range(1000):

    calibrated_gyro = calibrator.update(raw_gyro)
    angle += calibrated_gyro * dt

总结

核心理解：

  1. 零偏是固有的：制造工艺决定，无法完全消除
  2. 积分放大误差：小角速度误差 × 长时间 = 大角度误差
  3. 校准是关键：利用静止时段测量并补偿零偏
  4. 差分检测更可靠：关注变化量而非绝对值

这种校准方法在工程上非常实用，确保了陀螺仪在长时间运行中的角度精度！

1. MEMS陀螺仪如何测量角速度

MEMS陀螺仪，全称为微机电系统（Micro-Electro-Mechanical System）陀螺仪，是一种利用微机电系统技术制造的角速度传感器。它能够测量物体围绕某个轴旋转的角速度，广泛应用于汽车、航空航天、消费电子、工业自动化等领域。MEMS陀螺仪通常基于科里奥利效应（Coriolis Effect）工作。当一个物体在旋转的参考系中移动时，它会感受到一个垂直于其运动方向和旋转轴的力，这个力就是科里奥利力。MEMS陀螺仪通过检测这个力来测量角速度。

MEMS技术在机器人领域中扮演着“感官”和“小脑”的角色，是机器人感知环境和实现精准运动控制的基础。比如MEMS惯性测量单元（IMU）：包含加速度计和陀螺仪，是机器人的“前庭系统”，提供自身的姿态、加速度、角速度信息，对于平衡控制、步态规划、精准定位至关重要。MEMS技术在其中扮演着“感官”和“小脑”的角色，是机器人感知环境和实现精准运动控制的基础。

class MEMSGyroscope:
    def measure_angular_velocity(self):
        # 对于每个轴：
        results = {}
        for axis in ['x', 'y', 'z']:
            # 1. 驱动质量块振动
            drive_signal = self.generate_drive_vibration(axis)
            
            # 2. 如果有旋转，科里奥利力使检测质量块位移
            coriolis_force = self.calculate_coriolis_force(axis, drive_signal)
            
            # 3. 测量电容变化
            capacitance_change = self.measure_capacitance_change(axis)
            
            # 4. 转换为角速度
            angular_velocity = self.convert_to_angular_velocity(capacitance_change)
            results[axis] = angular_velocity
            
        return results

def acceleration_effect(true_omega, acceleration):
    """
    加速度引起的误差
    true_omega: 真实角速度
    acceleration: 线性加速度 (m/s²)
    """
    # 加速度会使质量块产生额外位移
    error = k_accel * acceleration  # k_accel是加速度敏感系数
    return true_omega + error

def cross_axis_coupling(omega_true):
    """
    交叉轴误差模型
    """
    omega_measured = [
        omega_true[0] + k_xy * omega_true[1] + k_xz * omega_true[2],
        omega_true[1] + k_yx * omega_true[0] + k_yz * omega_true[2], 
        omega_true[2] + k_zx * omega_true[0] + k_zy * omega_true[1]
    ]
    return omega_measured

class AdvancedIMU:
    def __init__(self):
        self.gyro_bias = [0, 0, 0]
        self.accel_sensitivity = [0.1, 0.1, 0.1]  # 加速度敏感系数
        self.cross_axis_matrix = np.eye(3)  # 交叉轴补偿矩阵
        
    def compensated_measurement(self, raw_gyro, raw_accel):
        # 1. 补偿零偏
        gyro_no_bias = raw_gyro - self.gyro_bias
        
        # 2. 补偿加速度影响
        accel_compensation = np.dot(self.accel_sensitivity, raw_accel)
        gyro_accel_comp = gyro_no_bias - accel_compensation
        
        # 3. 补偿交叉轴影响
        final_gyro = np.dot(self.cross_axis_matrix, gyro_accel_comp)
        
        return final_gyro

    • 加速度敏感性：线性加速度会产生虚假角速度读数
    • 振动干扰：外部振动会干扰内部振动系统
    • 交叉轴耦合：一轴旋转会影响其他轴测量

  3. 解决方案：

    • 工厂校准：预先标定各种误差参数
    • 传感器融合：结合加速度计、磁力计等校正陀螺仪
    • 自适应算法：运行时持续校准和补偿

这就是为什么高端IMU价格昂贵——它们包含了精密的校准和补偿算法来应对各种运动状态下的测量挑战！
