import "strings"
// 任务配置：每秒自动运行一次
option task = {name: "DynamicElectricitySum", every: 1s}
// 定义用于筛选 measurement 的关键字
keyword = "电度每小时"

// 1. 查询近一天内所有的 _measurement 字段（即 measurement 名称）并去重
all_measurements = from(bucket: "history")
    |> range(start: -1d) // 查询最近1小时的数据，防止历史太久没数据的 measurement 被漏掉
    |> keep(columns: ["_measurement"]) // 只保留 _measurement 列
    |> distinct(column: "_measurement") // 去重，得到所有出现过的 measurement 名称
// 2. 用 map 生成一个带 _measurement 字符串的表，方便筛选
filtered_measurements = all_measurements
    |> filter(fn: (r) => strings.containsStr(v: r._measurement, substr: keyword)) // 筛选 measurement 名称里包含关键字的项
    |> keep(columns: ["_measurement"]) // 只保留 _measurement 列
// 3. 收集所有符合条件的 measurement 名称到一个数组
names_record =
    filtered_measurements
        |> group(columns: [])                      // 合并为单表，避免分组导致 reduce 拆分
        |> reduce(
            // 注意 reduce 的形参名必须是 (r, accumulator)
            fn: (r, accumulator) => ({
                // 字符串累加（避免数组 + 号报错）
                names: if accumulator.names == "" then r._measurement else accumulator.names + "|" + r._measurement,
            }),
            identity: {names: ""},                 // 初始为空串
        )
        |> findRecord(fn: (key) => true, idx: 0)   // 取唯一（或第一）条记录
// 把拼接串切分为数组；若为空串则给空数组
measurement_names =
    if names_record.names == "" then [] else strings.split(v: names_record.names, t: "|")
// 4. 查询近1秒内所有 measurement 的数据，并只保留 measurement 名称在 measurement_names 里的数据
from(bucket: "history")
    |> range(start: -1s) // 只查最近1秒的数据
    |> filter(fn: (r) => contains(value: r._measurement, set: measurement_names)) // 只保留 measurement 名称在 measurement_names 里的行
    |> filter(fn: (r) => r._field == "v") // 只处理 _field 为 v 的数据
    |> aggregateWindow(every: 1s, fn: sum, createEmpty: false) // 以1秒为窗口做 sum 聚合
    |> map(
        fn: (r) => ({
            _time: r._time, // 保留时间戳

            _value: r._value, // 保留聚合后的值

            // measurement 名称替换，电度每小时 => 电度每秒

            _measurement: "总电度",

        })
    )
    |> to(
        bucket: "history", // 写回的 bucket
        org: "uzone", // 组织
        measurementColumn: "_measurement", // 指定 measurement 字段
        fieldFn: (r) => ({"v": r._value, "q": 192}), // 写入的字段
    )

import "experimental"
from(bucket: "history")
  |> range(start: v.timeRangeStart, stop: v.timeRangeStop)
  |> filter(fn: (r) => r["_measurement"] == "降水现象仪/当前降水现象")
  |> filter(fn: (r) => r["_field"] == "v")
  |> map(fn: (r) => ({
      r with
      _time: experimental.addDuration(d: 8h, to: r._time)
  }))
