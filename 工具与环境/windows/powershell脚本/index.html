$services = @("WpcMonSvc", "RetailDemo")

foreach ($service in $services) {
    # 停止服务
    Get-Service -Name $service | Stop-Service -Force
    
    # 设置服务的启动类型为禁用
    Set-Service -Name $service -StartupType Disabled
}
