Vim

vim的全局配置一般修改/etc/vim/vimrc或者/etc/vimrc，对所有用户生效，修改~/.vimrc仅对个人用户生效，或者在单词编辑中可以直接在命令行中写set number这种命令生效

Window下vim的配置文件名为_vimrc，位于vim安装目录下

-- 开启语法高亮
vim.o.syntax = "enable"
-- 显示相对行号
vim.o.relativenumber = true
-- 显示行号
vim.o.number = true
-- 高亮所在行
vim.o.cursorline = true
-- 自动换行
vim.o.wrap = true
-- 显示光标位置
vim.o.ruler = true
-- 边输入边搜索
vim.o.incsearch = true
-- 开启搜索匹配高亮
vim.o.hlsearch = true
-- 搜索时自行判断是否需要忽略大小写
vim.o.smartcase = true

-- tab键转换为 4 个空格
vim.o.tabstop = 4
vim.o.softtabstop = 4
vim.o.shiftwidth = 4
-- 新行对齐当前行，tab转换为空格
vim.o.expandtab = true
vim.bo.expandtab = true
vim.o.autoindent = true
vim.bo.autoindent = true
vim.o.smartindent = true

-- << >> 缩进时移动的长度
vim.o.shiftwidth = 4
vim.bo.shiftwidth = 4

-- 使用jk移动光标时，上下方保留8行
vim.o.scrolloff = 8
vim.o.sidescrolloff = 8

-- 设置自动折叠
vim.o.smartindent = true
-- 历史命令最多保存1000条
vim.o.history = 1000
-- 显示空白字符
vim.o.list = true
-- 样式
vim.o.background = "dark"
vim.o.termguicolors = true
vim.opt.termguicolors = true

上述代码中，vim.o.syntax = "enable" 和syntax on的区别是，

前者使用了lua语法来设置vim的语法高亮选项，可以自定义颜色；后者用vimscript语言来设置vim的语法高亮，用vim默认的颜色。

-- ~/.config/nvim/init.lua

vim.opt.number         = true  -- 显示绝对行号
vim.opt.relativenumber = true  -- 同时显示相对行号（跳转更方便）

-- 语法高亮（Neovim 0.9+ 已内置 treesitter，下面会再补一份更现代的）
vim.cmd('syntax on')

vim.opt.cursorline     = true  -- 高亮当前屏幕行，眼睛不易花

vim.opt.tabstop        = 4     -- 按 Tab 时屏幕上显示的宽度（字符数）
vim.opt.shiftwidth     = 4     -- 自动缩进宽度（<< >> ==）
vim.opt.expandtab      = true  -- 把真正的 Tab 字符替换成空格，避免合作时冲突

vim.opt.ignorecase     = true  -- 搜索默认忽略大小写
vim.opt.smartcase      = true  -- 一旦搜索词里出现大写字母，则临时变成区分大小写
vim.opt.hlsearch       = true  -- 把上一次搜索结果继续高亮（可用 <leader><Esc> 清屏）

vim.opt.showcmd        = true  -- 右下角显示正在输入的命令（可视化反馈）
vim.opt.wildmenu       = true  -- 命令行 Tab 补全出现“菜单条”而非单纯循环
vim.opt.autoindent     = true  -- 回车后继承上一行缩进
vim.opt.clipboard      = "unnamedplus"  -- 把系统剪贴板当作默认寄存器（需要外部 provider）

-----------------------------------------------
---
-- 2. 强烈建议再补的“通用 20 条”
-----------------------------------------------
---
vim.opt.mouse          = 'a'           -- 全模式启用鼠标（选/拖/滚轮）
vim.opt.signcolumn     = 'yes'         -- 左侧诊断栏常驻，避免窗口左右晃动
vim.opt.wrap           = false         -- 长行不折行，用 <lj> 左右移
vim.opt.scrolloff      = 8             -- 光标上下最少留 8 行，阅读更舒适
vim.opt.sidescrolloff  = 8             -- 左右同理
vim.opt.splitbelow     = true          -- 水平分割新窗口默认放下边
vim.opt.splitright     = true          -- 垂直分割新窗口默认放右边
vim.opt.termguicolors  = true          -- 启用 24 位真彩（主题必备）
vim.opt.updatetime     = 250           -- 文件变动检测 / swap 写入频率(ms)
vim.opt.timeoutlen     = 500           -- 键序列超时（映射 <leader> 系列时手感好）
vim.opt.incsearch      = true          -- 增量搜索，边打边跳
vim.opt.hidden         = true          -- 允许不保存就切换 buffer
vim.opt.backup         = false         -- 不保留 *.bak
vim.opt.writebackup    = false         -- 写入时不生成备份
vim.opt.swapfile       = false         -- 不产生 .swp（用 git 管理即可）
vim.opt.undofile       = true          -- 永久撤销（~/.local/share/nvim/undo）
vim.opt.undodir        = os.getenv("HOME") .. '/.local/share/nvim/undo'
vim.opt.list           = true          -- 显示不可见字符
vim.opt.listchars      = 'tab:▸ ,trail:·,nbsp:+'  -- 把空格/制表符/换行符画出来
vim.opt.completeopt    = 'menu,menuone,noselect'  -- 内置补全体验更友好

-----------------------------------------------
---
-- 3. 懒人键位（示例）
-----------------------------------------------
---
vim.g.mapleader        = ' '           -- 把 <leader> 设成空格，按起来快
vim.g.maplocalleader   = '\\'

-- 保存 / 退出
vim.keymap.set('n', '<leader>w', '<Cmd>w<CR>')
vim.keymap.set('n', '<leader>q', '<Cmd>q<CR>')

-- 快速清搜索高亮
vim.keymap.set('n', '<leader><Esc>', '<Cmd>nohlsearch<CR>')

-- 新建文件自动插入模板、或禁止自动注释换行
vim.api.nvim_create_autocmd("FileType", {
  pattern = "*",
  callback = function()
    vim.opt_local.formatoptions:remove({ "c", "r", "o" })
  end,
})
