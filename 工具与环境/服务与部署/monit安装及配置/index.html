监控一个python Web应用

# 在 conf.d/ 目录创建 webapp.conf
check process webapp with pidfile /var/run/webapp.pid
    start program = "/usr/bin/systemctl start webapp"
    stop program = "/usr/bin/systemctl stop webapp"
    if failed host 127.0.0.1 port 5000 protocol http
        and request "/health" for 3 cycles then restart
    if cpu > 90% for 5 cycles then restart
    if 3 restarts within 5 cycles then timeout
    group webapp

#TestMonit 应用监控配置
check process TestMonit matching "TestMonit.dll"
    start program = "/usr/local/bin/testmonit_service.sh start"
      with timeout 30 seconds
    stop program = "/usr/local/bin/testmonit_service.sh stop"
      with timeout 15 seconds

    # HTTP 健康检查
    if failed host 127.0.0.1 port 5000 protocol http
        request "/todos/ping" with content = "pong"
        with timeout 10 seconds
        for 2 cycles
        then restart
    # 重启策略
    if 3 restarts within 5 cycles then timeout
