#通过官方提供的安装脚本安装
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh
#运行脚本之后交互式安装即可
bash Miniconda3-latest-Linux-x86_64.sh
#注意：如果安装的时候选择了no，conda命令是默认不会被添加到环境变量里的，要手动处理一下

# 1. 为项目创建独立环境
conda create -n web_app python=3.11
conda activate web_app

# 2. 安装项目依赖
conda install django pandas
pip install some_special_package  # 也可以混用pip

# 3. 工作完成后退出环境
conda deactivate

# 4. 环境导出（用于部署或分享）
conda env export > environment.yml

# 5. 从文件重建环境
conda env create -f environment.yml

使用

查看有哪些虚拟环境 

1

conda env list

激活特定虚拟环境 

1

conda activate xxx

关闭虚拟环境 

1

conda deactivate

# 1. 安装 conda-pack 工具(需要在你想迁移的环境中运行)
conda install -c conda-forge conda-pack
# 2. 打包 rms 环境
# 这会在当前目录生成一个 rms.tar.gz 文件
conda pack -n rms -o rms.tar.gz
# （可选）查看包大小
ls -sh rms.tar.gz
# 也可以直接指定安装的虚拟环境
conda install -n rms -c conda-forge conda-pack
